**Добавить endpoints:**

```python
@router.post("/{video_id}/run")
async def run_workflow(video_id: int):
    """AUTO mode: запустить все шаги."""
    return await orchestrator.run_auto(video_id)

@router.post("/{video_id}/generate/{step}")
async def generate_step(video_id: int, step: str, feedback: str = None):
    """Генерировать шаг (MANUAL mode)."""
    return await orchestrator.generate_step(video_id, step, feedback)

@router.get("/{video_id}/history/{step}")
async def get_history(video_id: int, step: str):
    """Получить ВСЕ варианты для шага."""
    return await orchestrator.get_step_history(video_id, step)

@router.post("/{video_id}/select/{variant_id}")
async def select_variant(video_id: int, variant_id: int):
    """Выбрать вариант и auto-continue."""
    return await orchestrator.select_variant(video_id, variant_id)
```
